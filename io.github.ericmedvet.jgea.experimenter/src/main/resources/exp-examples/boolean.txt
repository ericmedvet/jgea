$basePath = ''../../Documenti/experiments/{^^.name}/{^^.startTime}/''

ea.experiment(
  runs = (randomGenerator = (seed = [1:1:1]) * [m.defaultRG()])
    * (problem = [
      ea.p.br.mopm(n = 2);
      ea.p.br.mopm(n = 3);
      ea.p.br.evenParity(n = 4);
      ea.p.br.targetBased(
        name = "xor-not-xor";
        n = 2;
        exprs = [
          "and(or(0;1);not(and(0;1)))";
          "not(and(or(0;1);not(and(0;1))))"
        ]
      )
    ])
    * (solver = (stop = ea.sc.nOfQualityEvaluations(n = 500)) * [
      ea.s.ga(
        name = "ge";
        representation = ea.r.bitString();
        mapper = ea.m.multiBTreeToBf(of = ea.m.cfgTreeToMultiBTree(
          of = ea.m.bsToGeCfgTree(grammar = ea.grammar.string.booleanRegression())
        ))
      );
      ea.s.ga(
        name = "hge";
        representation = ea.r.bitString();
        mapper = ea.m.multiBTreeToBf(of = ea.m.cfgTreeToMultiBTree(
          of = ea.m.bsToHgeCfgTree(grammar = ea.grammar.string.booleanRegression(); length = 128)
        ))
      );
      ea.s.ga(
        name = "whge";
        representation = ea.r.bitString();
        mapper = ea.m.multiBTreeToBf(of = ea.m.cfgTreeToMultiBTree(
          of = ea.m.bsToWhgeCfgTree(grammar = ea.grammar.string.booleanRegression(); length = 128)
        ))
      );
      ea.s.ga(
        name = "gp";
        mapper = ea.m.multiBTreeToBf();
        representation = ea.r.multiBTree()
      );
      ea.s.ga(
        name = "cfggp";
        representation = ea.r.cfgTree(grammar = ea.grammar.string.booleanRegression());
        mapper = ea.m.multiBTreeToBf(of = ea.m.cfgTreeToMultiBTree())
      )
    ])
    * [ea.run()];
  listeners = [
    ea.l.console(
      eFunctions = [
        ea.f.size(of = ea.f.genotype(of = ea.f.best()); format = "%4d");
        ea.f.size(of = ea.f.solution(of = ea.f.best()); format = "%3d");
        f.mapValue(key = avg_dissimilarity; of = ea.f.quality(of = ea.f.best()); format = "%6.2f");
        ea.f.hist(of = f.each(of = ea.f.all(); mapF = f.mapValue(key = avg_dissimilarity; of = ea.f.quality())));
        f.timestamp()
      ];
      onlyLast = true
    );

    ea.l.individualCsv(
      path = $basePath + "all.csv";
      splitterF = ea.f.firsts();
      onlyLast = true;
      ieFunctions = [
        ea.f.size(of = ea.f.genotype(); format = "%4d");
        ea.f.size(of = ea.f.solution(); format = "%3d");
        f.mapValue(key = avg_dissimilarity; of = ea.f.quality(); format = "%6.2f")
      ]
    );

    % ========= CONVERGENCE PLOTS
    ea.l.savePlotForExp(
      path = $basePath + "best-fitness.svg";
      plot = ea.plot.multi.quality(q = f.mapValue(key = avg_dissimilarity));
      type = svg
    );

    % ========= BOX PLOTS
    ea.l.savePlotForExp(
      path = $basePath + "final-best-fitness.svg";
      plot = ea.plot.multi.qualityBoxplot(q = f.mapValue(key = avg_dissimilarity));
      configuration = viz.plot.conf.image(axesShow = all; axesIndependences = [all]);
      type = svg
    );

    % ========= TREES
    ea.l.saveForRun(
      path = $basePath + "{solver.name}/{problem.name}/best-{randomGenerator.seed:%02d}.svg";
      of = ea.acc.lastBest();
      processor = viz.f.toImage(
        drawer = viz.d.stacked(drawers = [
          viz.d.multi(drawer = ea.d.tree(scale = 2));
          viz.d.multi(drawer = viz.d.composition(
            drawer = ea.d.tree(scale = 2);
            f = ea.f.simplifiedBTree()
          ))
        ]);
        of = ea.f.genotype();
        type = svg
      );
      kCondition = predicate.matches(f = f.mappableKey(key = "solver.name"); regex = "gp")
    );
    ea.l.saveForRun(
      path = $basePath + "{solver.name}/{problem.name}/best-{randomGenerator.seed:%02d}.svg";
      of = ea.acc.lastBest();
      processor = viz.f.toImage(
        drawer = ea.d.tree(scale = 3);
        of = ea.f.genotype();
        type = svg
      );
      kCondition = predicate.matches(f = f.mappableKey(key = "solver.name"); regex = "cfggp")
    )
  ]
)
