$basePath = ''../../Documenti/experiments/{^^.name}/{^^.startTime}/''
$dT = 0.1
$tRange = m.range(min = 0; max = 30)
$qFunction = f.mapValue(key = "final.dist"; of = ea.f.behaviorQuality(); format = "%5.3f")
$sDescX = ds.e.n.x(of = ds.e.n.finalRobotP(of = ea.f.behavior(of = ea.f.quality())))
$sDescY = ds.e.n.y(of = ds.e.n.finalRobotP(of = ea.f.behavior(of = ea.f.quality())))
$descRangeX = m.range(min = 0; max = 1)
$descRangeY = m.range(min = 0; max = 1)
$nOfPoints = 200

$arenaE = ds.arena.prepared(
  name = empty;
  initialRobotXRange = m.range(min = 0.45; max = 0.5);
  initialRobotYRange = m.range(min = 0.78; max = 0.8);
  targetXRange = m.range(min = 0.3; max = 0.6)
)
$arenaB = ds.arena.prepared(
  name = a_barrier;
  initialRobotXRange = m.range(min = 0.45; max = 0.5);
  initialRobotYRange = m.range(min = 0.78; max = 0.8)
)
$arenaS = ds.arena.prepared(
  name = snake;
  initialRobotXRange = m.range(min = 0.1; max = 0.12);
  initialRobotYRange = m.range(min = 0.1; max = 0.12);
  targetXRange = m.range(min = 0.9; max = 0.95);
  targetYRange = m.range(min = 0.9; max = 0.95)
)

ea.experiment(
  runs = (randomGenerator = (seed = [1:1:1]) * [m.defaultRG()])
  * (solver = (stop = ea.sc.nOfQualityEvaluations(n = 5000))
  * (representation = [ea.representation.doubleString()])
  * (mapper = [
    ea.mapper.dsToParametrized(name = "mlp"; parametrized = ds.num.mlp(innerLayers = [8]))
  ])
  * [
    ea.s.ga(name = ''ga+{mapper.name}'');
    ea.s.mapElites(
      name = ''me-{archive.name}+{mapper.name}'';
      descriptors = [$sDescX; $sDescY];
      archive = ea.s.me.a.voronoi(range1 = $descRangeX; range2 = $descRangeY; nOfPoints = $nOfPoints)
    )
  ])
  * (problem = (simulation = (environment = (arena = [
    $arenaE;
    $arenaB;
    $arenaS
  ])
  * [ds.e.navigation()])
  * [ds.sat.fromEnvironment()])
  * [
    ea.p.simToSbbmo(
      name = ''{simulation.environment.arena.name}'';
      dT = $dT;
      tRange = $tRange;
      toMinObjectives = [ds.e.n.finalD()]
    )
  ])
  * [ea.run()];
  listeners = [
    ea.l.console(
      eFunctions = [
        f.composition(of = ea.f.quality(of = ea.f.best()); then = $qFunction);
        f.size(of = ea.f.genotype(of = ea.f.best()); format = "%3d")
      ];
      onlyLast = false
    );

    % ========= FITNESS PROGRESSION PLOTS
    ea.l.savePlotForExp(
      path = $basePath + "best-fitness.svg";
      plot = ea.plot.multi.quality(q = $qFunction);
      type = svg
    );

    % ========= ARCHIVE PLOTS
    ea.l.savePlotForRun(
      path = $basePath + "{solver.name}/{problem.name}/{randomGenerator.seed}-archive.svg";
      plot = ea.plot.single.me(q = $qFunction);
      kCondition = predicate.matches(f = f.mappableKey(key = "solver.name"); regex = "me.*");
      type = svg
    )
  ]
)
